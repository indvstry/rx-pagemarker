// SECTION 10: UI

var UI = (function() {
    var _editor = null;
    var _statsEl = null;
    var _zoomLevelEl = null;

    function init(editorEl, statsEl, zoomLevelEl) {
        _editor = editorEl;
        _statsEl = statsEl;
        _zoomLevelEl = zoomLevelEl;
    }

    function updateStats() {
        var markers = _editor.querySelectorAll('.page-marker');
        var count = markers.length;
        _statsEl.textContent = '';
        var strong = document.createElement('strong');
        strong.textContent = count;
        _statsEl.appendChild(strong);
        _statsEl.appendChild(document.createTextNode(' page marker' + (count !== 1 ? 's' : '')));
    }

    function setZoom(value) {
        var clamped = Math.max(50, Math.min(200, value));
        State.set('zoom', clamped);
        _editor.style.fontSize = clamped + '%';
        _zoomLevelEl.textContent = clamped + '%';
    }

    return { init: init, updateStats: updateStats, setZoom: setZoom };
})();

var FileLoader = (function() {
    var _editor = null;
    var _fileInput = null;

    function init(editorEl, fileInputEl) {
        _editor = editorEl;
        _fileInput = fileInputEl;
    }

    function loadContent(html) {
        // adapt from existing loadContent()
        // Replace: history = [...]; historyIndex = 0;
        //   with: History.reset(_editor);
        // Replace: setupMarkerDragging()
        //   with: Markers.setupListeners(_editor);
        // Replace: updateButtons(), updateStats()
        //   with: UI.updateStats()
        //   (History.reset handles button state)
    }

    function _attachFileInput() {
        _fileInput.addEventListener('change', function(e) {
            var file = e.target.files[0];
            if (!file) return;
            var reader = new FileReader();
            reader.onload = function(ev) {
                State.set('originalHTML', ev.target.result);
                State.set('originalFileName', file.name);
                loadContent(State.originalHTML);
                EventBus.emit('file:loaded', { html: State.originalHTML, fileName: State.originalFileName });
                Toast.show('File loaded successfully', 'success');
            };
            reader.onerror = function() { Toast.show('Error loading file', 'error'); };
            reader.readAsText(file);
        });
    }

    // Subscribe History.reset to file:loaded so history is always reset on new file
    EventBus.on('file:loaded', function(data) {
        History.reset(_editor);
        DragDrop.init(_editor);
        AddMode.init(_editor);
    });

    return { init: init, loadContent: loadContent, attachFileInput: _attachFileInput };
})();
