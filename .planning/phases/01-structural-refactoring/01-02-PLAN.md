History IIFE (complete structure — adapt bodies from existing flat-scope functions):
```javascript
// ============================================================
// SECTION 4: History
// ============================================================
var History = (function() {
    var stack = [];
    var index = -1;
    var MAX = 50;
    var _editor = null;

    function _capture() {
        if (index < stack.length - 1) { stack = stack.slice(0, index + 1); }
        stack.push(_editor.cloneNode(true));
        index = stack.length - 1;
        if (stack.length > MAX) { stack.shift(); index--; }
        State.set('hasUnsavedChanges', true);
        AutoSave.save();
        _updateButtons();
        UI.updateStats();
    }

    function _restore(savedEditor) {
        _editor.textContent = '';
        Array.from(savedEditor.childNodes).forEach(function(node) {
            _editor.appendChild(node.cloneNode(true));
        });
        _editor.classList.remove('empty');
        _updateButtons();
        UI.updateStats();
        EventBus.emit('history:restored');  // AFTER DOM is fully rebuilt
    }

    function _updateButtons() {
        document.getElementById('undoBtn').disabled = index <= 0;
        document.getElementById('redoBtn').disabled = index >= stack.length - 1;
    }

    // Subscribe to all mutation events — _capture() is ONLY called via EventBus
    EventBus.on('markers:moved',   _capture);
    EventBus.on('markers:added',   _capture);
    EventBus.on('markers:deleted', _capture);
    EventBus.on('markers:updated', _capture);

    return {
        undo: function() { if (index > 0) { index--; _restore(stack[index]); } },
        redo: function() { if (index < stack.length - 1) { index++; _restore(stack[index]); } },
        reset: function(editorEl) {
            _editor = editorEl;
            stack = [_editor.cloneNode(true)];
            index = 0;
            _updateButtons();
        }
    };
})();
```

NOTE: `_capture()` calls `AutoSave.save()` and `UI.updateStats()`. AutoSave is defined AFTER History in load order — this is fine because _capture() is called at runtime (not at parse time).     var failCount = 0;
    var MAX_FAILS = 3;

    function save() {
        var editor = document.getElementById('editor');
        try {
            var data = {
                originalHTML: State.originalHTML,
                editorContent: editor.innerHTML,
                timestamp: Date.now()
            };
            localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
            failCount = 0;
        } catch (e) {
            failCount++;
            console.warn('[AutoSave] Failed to save:', e.message);

                Toast.show('Auto-save disabled: ' + e.message + '. Please download your work manually!', 'error');
            }
        }
    }

    function clear() {
        try {
            localStorage.removeItem(STORAGE_KEY);
        } catch (e) {
            console.warn('[AutoSave] Failed to clear saved state:', e.message);
        }
    }
